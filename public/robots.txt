# Content Signals (contentsignals.org, draft-romm-aipref-contentsignals)
# Stride's preferences for what may be done with the pages a crawler is
# allowed to fetch:
#
#   search=yes    Index this content and link to it in search results.
#   ai-input=yes  Use it to ground AI answers — retrieval, summaries, citations.
#                 Deliberately yes: /llms.txt, the markdown twin of every page,
#                 the MCP servers and /ask all exist so assistants CAN answer
#                 questions about the club accurately.
#   ai-train=no   Do not use it as model training data. Grounding an answer
#                 about Stride cites Stride; baking this year's event prices
#                 into a model's weights does not, and cannot be corrected when
#                 they change.
#
# These are usage preferences, not access control. They say nothing about the
# Disallow rules below, which still apply — a signal of "yes" never opens a
# path a crawler was told not to fetch.

User-agent: *
Content-Signal: search=yes, ai-input=yes, ai-train=no
Allow: /
Disallow: /admin/
Disallow: /api/
Disallow: /auth/
Disallow: /events/*/confirmation/
Disallow: /my-runs
Disallow: /login/
Disallow: /register/
